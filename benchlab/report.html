```python
    m, n = len(a), len(b)
    
    # dp[i][j] holds the edit distance between a[:i] and b[:j]
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    # Base cases: transforming to/from an empty string
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
        
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                # Characters match: no extra cost
                dp[i][j] = dp[i - 1][j - 1]
```

```python
def min_edit_distance(a: str, b: str) -> int:
    """Return the Levenshtein edit distance between strings a and b.

    Each operation (insertion, deletion, or substitution) costs 1.
    Uses dynamic programming with O(len(a) * len(b)) time and space.
    """
    m, n = len(a), len(b)

    # dp[i][j] = edit distance between a[:i] and b[:j]
    dp = [[0] * (n + 1) for _ in range(m + 1)]

    # Base cases: transforming to/from an empty string
    for i in range(m + 1):
        dp[i][0] = i   # delete all characters from a[:i]
    for j in range(n + 1):
        dp[0][j] = j   # insert all characters into b[:j]

    # Fill the DP table
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]           # no change needed
            else:
                dp[i][j] = 1 + min(
                    dp[i - 1][j],    # deletion from a
                    dp[i][j - 1],    # insertion into b
                    dp[i - 1][j - 1] # substitution
                )

    return dp[m][n]
```

```python
def lis_length(nums):
    """Return the length of the longest strictly increasing subsequence in nums."""
    if not nums:
        return 0

    # tails[i] = smallest tail element for an increasing subsequence of length i+1
    tails = []

    import bisect
    for num in nums:
        idx = bisect.bisect_left(tails, num)
        if idx == len(tails):
            tails.append(num)
        else:
            tails[idx] = num

    return len(tails)
```

```python
def spiral_order(matrix):
    """Return the elements of a 2D matrix in clockwise spiral order starting from top-left."""
    if not matrix or not matrix[0]:
        return []

    result = []
    top, bottom = 0, len(matrix) - 1
    left, right = 0, len(matrix[0]) - 1

    while top <= bottom and left <= right:
        # Traverse the top row from left to right
        for i in range(left, right + 1):
            result.append(matrix[top][i])
        top += 1

        # Traverse the right column from top to bottom
        for i in range(top, bottom + 1):
            result.append(matrix[i][right])
        right -= 1

        # Traverse the bottom row from right to left (if still valid)
        if top <= bottom:
            for i in range(right, left - 1, -1):
                result.append(matrix[bottom][i])
            bottom -= 1

        # Traverse the left column from bottom to top (if still valid)
        if left <= right:
            for i in range(bottom, left - 1, -1):
                result.append(matrix[i][left])
            left += 1
```

```python
    stack = []
    for token in tokens:
        if token == '+':
            b, a = stack.pop(), stack.pop()
            stack.append(a + b)
        elif token == '-':

            stack.append(a - b)
```
